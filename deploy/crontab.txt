# ==============================================================================
# NSE Quant Lab - Production Crontab Schedule (Asia/Kolkata IST)
# Note: Ensure system timezone is set to Asia/Kolkata (`timedatectl set-timezone Asia/Kolkata`)
# All scripts internally verify `is_trading_day()` to automatically skip weekends & holidays.
# ==============================================================================
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
CRON_TZ=Asia/Kolkata

# 1. 08:30 IST: Morning Heartbeat & Status Check (Mon-Fri)
30 08 * * 1-5 cd %APP_DIR% && ./venv/bin/python scripts/run_scheduler_daemon.py --test-job morning_heartbeat >> logs/cron.log 2>&1

# 2. 09:16 IST: Execute Morning Market Open Orders (Mon-Fri)
16 09 * * 1-5 cd %APP_DIR% && ./venv/bin/python scripts/execute_paper_orders.py >> logs/cron.log 2>&1

# 3. 15:35 IST: Ingest Today's EOD Daily Candles from Upstox (Mon-Fri)
35 15 * * 1-5 cd %APP_DIR% && ./venv/bin/python scripts/ingest_daily_upstox_candles.py >> logs/cron.log 2>&1

# 4. 15:45 IST: Compute Indicators, Run Daily Strategy Pipeline & Telegram Signals (Mon-Fri)
45 15 * * 1-5 cd %APP_DIR% && ./venv/bin/python scripts/run_daily_pipeline.py >> logs/cron.log 2>&1
