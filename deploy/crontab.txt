# ==============================================================================
# NSE Quant Lab - Production Crontab Schedule (Asia/Kolkata IST)
# Zero-Spam Policy: Completely silent on weekends, holidays, and zero-trade days.
# Alerts are emitted strictly for real order fills, new signals, or system errors.
# ==============================================================================
SHELL=/bin/bash
PATH=/usr/local/sbin:/usr/local/bin:/usr/sbin:/usr/bin:/sbin:/bin
CRON_TZ=Asia/Kolkata

# 1. 09:16 IST: Execute Morning Market Open Orders (Mon-Fri, Trading Days Only)
16 09 * * 1-5 cd %APP_DIR% && ./venv/bin/python scripts/execute_paper_orders.py >> logs/cron.log 2>&1

# 2. 15:35 IST: Ingest Today's EOD Daily Candles from Upstox (Mon-Fri, Trading Days Only)
35 15 * * 1-5 cd %APP_DIR% && ./venv/bin/python scripts/ingest_daily_upstox_candles.py >> logs/cron.log 2>&1

# 3. 15:45 IST: Compute Indicators, Run Daily Strategy Pipeline & Telegram Signals (Mon-Fri, Trading Days Only)
45 15 * * 1-5 cd %APP_DIR% && ./venv/bin/python scripts/run_daily_pipeline.py >> logs/cron.log 2>&1
